-- Fixes the 3-way id collision on 'kb201'. Gives each of the 3 items its -- own unique id (kb201_1, kb201_2, kb201_3) — nothing else about them -- (title, content, category, status) is touched. Every other item in the -- knowledge base is left completely untouched. update kb_data set data = jsonb_set( data, '{knowledgeItems}', ( select jsonb_agg( case when elem->>'id' = 'kb201' then jsonb_set( elem, '{id}', to_jsonb('kb201_' || row_number() over (partition by (elem->>'id') order by elem->'en'->>'title')) ) else elem end order by ord ) from jsonb_array_elements(data->'knowledgeItems') with ordinality as t(elem, ord) ) ) where id = 1;
